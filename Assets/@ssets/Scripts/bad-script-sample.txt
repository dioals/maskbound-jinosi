// =====================================================================================
// BadConventionExample.cs
//
// PURPOSE: This script intentionally violates *every* rule from
// "Contoh Code Convention" (gametutorialid.blogspot.com). It's meant to be used
// as a "spot the violations" teaching sample for new programmers on the team -
// NOT as real production code.
//
// Every violation is flagged with a "// VIOLATION:" comment referencing the rule.
// =====================================================================================

using UnityEngine;
using System;
using System.Linq;
using System.Collections;
using System.Collections.Generic;

// VIOLATION (Namespaces): should start with "Toge.ProjectName", Pascal Case.
// Here it's lowercase and unrelated to the project.
namespace stuff
{
    // VIOLATION (Interface): should have "I" prefix, be an Adjective, Pascal Case.
    // Here: no prefix, it's a noun, lowercase.
    public interface damageable
    {
        // VIOLATION (Methods): should be Pascal Case verb -> should be "TakeDamage".
        void takeDamage();
    }

    // VIOLATION (Structs): should be Pascal Case noun -> should be "AmmoData".
    public struct ammoData
    {
        // VIOLATION (Fields/Locals): single-character names outside of loop 'i'.
        // VIOLATION (Multiple statements): two declarations crammed on one line.
        public int a, d;
    }

    // VIOLATION (Enums): name should be Pascal Case, values SCREAMING_CAPS.
    // Here: name and values are all lowercase, no explicit numeric values.
    public enum wpntype
    {
        pistol,
        rifle,
        smg
    }

    // VIOLATION (ScriptableObject Classes): should use suffix "SO" -> "WeaponDataSO".
    [CreateAssetMenu]
    public class weapondata : ScriptableObject
    {
        // VIOLATION (Fields): should be private camelCase with underscore,
        // exposed via a public Pascal Case property instead.
        // VIOLATION (Abbreviations): "wpnName" / "dmg" instead of full words.
        public string wpnName;
        public int dmg;
    }

    // VIOLATION (Abstract Classes): should be Pascal Case noun -> "ShootableWeapon".
    public abstract class shootThing : MonoBehaviour
    {
        public abstract void fire();
    }

    // VIOLATION (Extension Classes): should be static, suffix "Exts" -> "TransformExts".
    // VIOLATION: extension method missing "this" keyword, so it isn't even an extension.
    public class transformHelper
    {
        public void jump(Transform t, float h) { }
    }

    // VIOLATION (Classes): should be Pascal Case noun -> "Pistol".
    // Here: lowercase, abbreviated, verb-like.
    public class pStl : shootThing, damageable
    {
        // VIOLATION (Region naming): should be lower case region names, e.g. "fields".
        #region FIELDS

        // VIOLATION (Fields are always private): these are public with no [SerializeField].
        // VIOLATION (Fields naming): Pascal Case with no underscore instead of _camelCase.
        public int Ammo;
        public GameObject Target;

        // VIOLATION (Multiple statements in a single line).
        public float Dmg = 10, Rng = 5;

        // VIOLATION: single-character field name that isn't a loop iterator.
        public int q;

        // VIOLATION (Constants): should be SCREAMING_CAPS -> "MAX_AMMO".
        public const int maxAmmo = 30;

        // VIOLATION (Static Readonly): should be SCREAMING_CAPS -> "POSITION_OFFSET".
        public static readonly Vector3 posOffset = Vector3.zero;

        // VIOLATION: TODO left in the codebase, never cleaned up.
        // TODO: remove this before shipping
        private int unusedVar;

        #endregion

        // VIOLATION (Properties): should be Pascal Case with a real backing field.
        // Here: camelCase name, and it just forwards to a public field directly.
        public int currentAmmo
        {
            get { return Ammo; }
            set { Ammo = value; }
        }

        // VIOLATION (Events): should prefer "Action", Pascal Case, named "{Method}Event".
        // Here: a raw delegate (not "event" keyword), lowercase, no "Event" suffix.
        public delegate void FireHandler();
        public FireHandler onFire;

        void Awake()
        {
            // VIOLATION (Avoid var): type is known and simple, should be explicit "int".
            var x = 10;

            // VIOLATION (Multiple statements in a single line).
            int i = 3; int k = 4;

            // VIOLATION (Incrementing): "i = i + 1" instead of "i += 1".
            i = i + 1;

            // VIOLATION (String interpolation): concatenation instead of $"...".
            Debug.Log("ammo: " + Ammo + " k:" + k + " x:" + x);

            // VIOLATION (Null operator stuff): manual null check instead of
            // "Target ??= new GameObject();" / "Target?.SetActive(true);".
            if (Target != null)
            {
                Target.SetActive(true);
            }
            else
            {
                Target = new GameObject();
            }
        }

        // VIOLATION (Unity event functions): empty Update/FixedUpdate should be deleted.
        void Update()
        {
        }

        void FixedUpdate()
        {
        }

        void LateUpdate()
        {
            // VIOLATION (Linq performance): heavy LINQ chain running every frame.
            var targets = FindObjectsOfType<GameObject>()
                .Where(g => g.activeSelf)
                .OrderBy(g => g.name)
                .ToList();

            // VIOLATION (Avoid lambda expressions if possible): unnecessary lambda.
            Action doStuff = () => { Debug.Log("stuff"); };
            doStuff();

            // VIOLATION (Loop declarations outside init field): combined counters
            // declared and incremented inline instead of one outside the loop.
            // VIOLATION: iterator names are not "i" / not meaningful ("z", "w").
            for (int z = 0, w = 0; z < 10; z++, w += 2)
            {
                // VIOLATION (Incrementing): "w = w + 2" instead of "w += 2".
                w = w + 2;
            }

            // VIOLATION (Foreach Iterators): should be camelCase -> "item".
            List<GameObject> Items = new List<GameObject>();
            foreach (GameObject Item in Items)
            {
                Debug.Log(Item.name);
            }
        }

        // VIOLATION (Methods): should be Pascal Case verb -> "LogAmmoCount".
        public void ammoCount()
        {
            Debug.Log(Ammo);
        }

        public override void fire()
        {
            try
            {
                // VIOLATION (Incrementing): "Ammo = Ammo - 1" instead of "Ammo -= 1".
                Ammo = Ammo - 1;
                onFire?.Invoke();
            }
            catch (Exception)
            {
                // VIOLATION: exception silently swallowed with no handling/logging.
            }
        }

        // VIOLATION (Interface method naming): should be Pascal Case -> "TakeDamage".
        public void takeDamage()
        {
            Debug.Log("ouch");
        }

        // VIOLATION (Coroutines): should be Pascal Case verb with "Do" prefix,
        // e.g. "DoFire". Here: camelCase, no prefix.
        public IEnumerator fireRoutine()
        {
            yield return null;
        }

        // VIOLATION (Special Methods): UI-callback methods should have "UI_" prefix.
        // This is wired up via drag-and-drop from a Button's OnClick() in the Inspector.
        public void ClickFireButton()
        {
            fire();
        }

        void someMethod()
        {
            // VIOLATION (Local Methods): should be Pascal Case verb -> "DoThing".
            void doThing()
            {
                Debug.Log("thing");
            }
            doThing();
        }

        // VIOLATION (Method Parameters): should be camelCase -> "amount", "duration".
        public void ReloadWeapon(int Amount, float Time)
        {
            Ammo += Amount;
        }

        // VIOLATION (Lambda Parameters): default should be "x", not "q".
        void lambdaExample()
        {
            List<int> nums = new List<int> { 1, 2, 3 };
            nums.RemoveAll(q => q == 0);
        }
    }

    // VIOLATION (Generic Parameter Types): should use short names like T, U
    // (or T{BaseClassName}), not full descriptive lowercase words.
    public class weaponManager<weapontype, weapondatatype> : MonoBehaviour
    {
    }

    // VIOLATION (Editor Classes): should be Pascal Case with suffix "Editor".
    public class pstlinspector
    {
    }
}
